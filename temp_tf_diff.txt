+export function getSharedPlanetTexture(
+  id: string,
+  realTextureUrls: Record<string, string>,
+  onUpdate?: () => void
+): THREE.Texture {
+  const fallbackTex = createProceduralTexture(id);
+  const state = textureLoadingStates.get(id);
+
+  if (state === 'loaded' || state === 'failed') {
+    return fallbackTex;
+  }
+
+  if (state === 'loading') {
+    if (onUpdate) {
+      if (!textureLoadCallbacks.has(id)) {
+        textureLoadCallbacks.set(id, []);
+      }
+      textureLoadCallbacks.get(id)!.push(onUpdate);
+    }
+    return fallbackTex;
+  }
+
+  const realUrl = realTextureUrls[id];
+  if (realUrl) {
+    textureLoadingStates.set(id, 'loading');
+    if (onUpdate) {
+      if (!textureLoadCallbacks.has(id)) {
+        textureLoadCallbacks.set(id, []);
+      }
+      textureLoadCallbacks.get(id)!.push(onUpdate);
+    }
+
+    sharedTextureLoader.load(
+      realUrl,
+      (loadedTex) => {
+        loadedTex.colorSpace = THREE.SRGBColorSpace;
+        loadedTex.wrapS = THREE.RepeatWrapping;
+        loadedTex.wrapT = THREE.RepeatWrapping;
+
+        const img = loadedTex.image;
+        const canvas = fallbackTex.image as HTMLCanvasElement;
+        if (canvas && canvas.getContext) {
+          canvas.width = img.width;
+          canvas.height = img.height;
+          const ctx = canvas.getContext('2d');
+          if (ctx) {
+            ctx.drawImage(img, 0, 0);
+            fallbackTex.wrapS = loadedTex.wrapS;
+            fallbackTex.wrapT = loadedTex.wrapT;
+            fallbackTex.colorSpace = loadedTex.colorSpace;
+            fallbackTex.needsUpdate = true;
+          }
+        }
+        textureLoadingStates.set(id, 'loaded');
+        const cbs = textureLoadCallbacks.get(id);
+        if (cbs) {
+          cbs.forEach(cb => {
+            try { cb(); } catch (e) { console.error(e); }
+          });

+          });
+          textureLoadCallbacks.delete(id);
+        }
+      }

+    textureLoadingStates.set(id, 'loaded');

+    rawTextureCallbacks.set(url, []);
+  }
+  rawTextureCallbacks.get(url)!.push(onLoad);
+
+  sharedTextureLoader.load(
+    url,
+    (tex) => {
+      tex.colorSpace = THREE.SRGBColorSpace;
+      rawTextureCache.set(url, tex);
+      rawTextureLoadingStates.set(url, 'loaded');
+      const cbs = rawTextureCallbacks.get(url);
+      if (cbs) {
+        cbs.forEach(cb => {
+          try { cb(tex); } catch (e) { console.error(e); }
+        });
+        rawTextureCallbacks.delete(url);
+      }
+    },
+    undefined,
+    (err) => {
+      console.warn(`Raw texture loading failed for ${url}`, err);
+      rawTextureLoadingStates.set(url, 'failed');
+      rawTextureCallbacks.delete(url);
+    }
+  );
+}
+
+export function clearSharedTextureCache(): void {
+  globalTextureCache.clear();
+  labelCache.clear();
+  textureLoadingStates.clear();
+  rawTextureCache.clear();
+  rawTextureLoadingStates.clear();
+  textureLoadCallbacks.clear();
+  rawTextureCallbacks.clear();
 }
